public class MarketEvaluationStrategy : BaseStrategy
{
    public override string Code => "S07";
    public override string Name => "赛道市场评估";
    public override StrategyType Type => StrategyType.Scoring;

    protected override StrategyResult ExecuteCore(ProductData product, ExecutionContext context)
    {
        var msi = CalculateMSI(product); // 市场规模指数
        var cii = CalculateCII(product); // 竞争强度指数
        var margin = CalculateMargin(product); // 毛利率
        var blueOcean = CalculateBlueOcean(product); // 蓝海指数
        
        var totalScore = (msi + cii + margin + blueOcean) / 4;
        
        return new StrategyResult
        {
            StrategyCode = Code,
            StrategyName = Name,
            Type = Type,
            Score = totalScore,
            Grade = GetGrade(totalScore),
            Decision = totalScore >= 70 ? "GO" : "WAIT",
            SubResults = new List<SubResult>
            {
                new SubResult { Name = "MSI", Score = msi },
                new SubResult { Name = "CII", Score = cii },
                new SubResult { Name = "毛利率", Score = margin },
                new SubResult { Name = "蓝海指数", Score = blueOcean }
            }
        };
    }
}

### S13 - 爆点识别引擎
```csharp
public class HotspotDetectionStrategy : BaseStrategy
{
    public override string Code => "S13";
    public override string Name => "爆点识别引擎";
    public override StrategyType Type => StrategyType.Detection;

    protected override StrategyResult ExecuteCore(ProductData product, ExecutionContext context)
    {
        var hotSignals = new List<string>();
        var declineWarnings = new List<string>();
        
        // 爆品信号检测 (12项)
        if (CheckSalesSpike(product)) hotSignals.Add("HOT-01: 销量突增");
        if (CheckSearchSpike(product)) hotSignals.Add("HOT-02: 搜索量暴涨");
        if (CheckBSRJump(product)) hotSignals.Add("HOT-04: BSR飙升");
        if (CheckReviewBurst(product)) hotSignals.Add("HOT-05: 好评井喷");
        
        // 衰退预警检测 (10项)
        if (CheckSalesDecline(product)) declineWarnings.Add("DEC-01: 销量持续下滑");
        if (CheckSearchDecline(product)) declineWarnings.Add("DEC-02: 搜索量萎缩");
        if (CheckBadReviews(product)) declineWarnings.Add("DEC-04: 差评爆发");
        
        var score = hotSignals.Count * 10 - declineWarnings.Count * 15;
        
        return new StrategyResult
        {
            StrategyCode = Code,
            StrategyName = Name,
            Type = Type,
            Score = Math.Max(Math.Min(score + 50, 100), 0),
            Warnings = declineWarnings,
            Suggestions = hotSignals
        };
    }
}
```

### S14 - 20节点决策树
```csharp
public class DecisionTreeStrategy : BaseStrategy
{
    public override string Code => "S14";
    public override string Name => "20节点决策树";
    public override StrategyType Type => StrategyType.Judgment;

    protected override StrategyResult ExecuteCore(ProductData product, ExecutionContext context)
    {
        var baseScore = 100m;
        var triggeredNodes = new List<string>();
        
        // 极高风险节点 (N01-N05) - 一票否决
        if (product.InfringementRisk == "高")
        {
            return new StrategyResult
            {
                StrategyCode = Code,
                StrategyName = Name,
                Type = Type,
                Score = 0,
                Decision = "STOP",
                Reason = "N01: 专利侵权风险 - 一票否决"
            };
        }
        
        if (product.PolicyRisk > 0.7m)
        {
            return new StrategyResult
            {
                StrategyCode = Code,
                StrategyName = Name,
                Type = Type,
                Score = 0,
                Decision = "STOP",
                Reason = "N02: 禁入类目 - 一票否决"
            };
        }
        
        // 高风险节点 (N06-N10)
        if (product.MonthlySearchVolume < 3000) { baseScore -= 30; triggeredNodes.Add("N06: 市场规模过小"); }
        if (product.TopConcentration > 0.6m) { baseScore -= 25; triggeredNodes.Add("N07: 竞争过于激烈"); }
        
        var margin = (product.TargetPrice - product.PurchaseCost) / product.TargetPrice;
        if (margin < 0.25m) { baseScore -= 30; triggeredNodes.Add("N08: 毛利率过低"); }
        
        // 中风险节点 (N11-N15)
        if (product.AverageRating < 4.2m) { baseScore -= 10; triggeredNodes.Add("N12: 评分偏低"); }
        if (product.ReturnRate > 0.1m) { baseScore -= 10; triggeredNodes.Add("N13: 退货风险"); }
        
        // 加分节点 (N16-N20)
        if (margin > 0.4m) { baseScore += 10; triggeredNodes.Add("N17: 高毛利"); }
        if (product.TopConcentration < 0.2m) { baseScore += 10; triggeredNodes.Add("N18: 低竞争"); }
        if (product.SearchGrowthRate > 0.3m) { baseScore += 10; triggeredNodes.Add("N19: 高增长"); }
        
        return new StrategyResult
        {
            StrategyCode = Code,
            StrategyName = Name,
            Type = Type,
            Score = baseScore,
            Grade = GetGrade(baseScore),
            Decision = baseScore >= 80 ? "GO" : baseScore >= 60 ? "WAIT" : "STOP",
            Reason = $"决策树评分: {baseScore:F1}分，触发{triggeredNodes.Count}个节点",
            Warnings = triggeredNodes.Where(n => n.Contains("风险") || n.Contains("过低")).ToList(),
            Suggestions = triggeredNodes.Where(n => n.Contains("优势") || n.Contains("高")).ToList()
        };
    }
}
```
