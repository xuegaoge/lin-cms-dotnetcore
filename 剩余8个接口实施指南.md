### 1. BI监控Service (BIDashboardService.cs)

```csharp
using AutoMapper;
using FreeSql;
using IGeekFan.FreeKit.Extras.FreeSql;
using LinCms.Application.Contracts.Selection;
using LinCms.Entities.Selection;
using System.Collections.Generic;
using System.Linq;
using System.Threading.Tasks;

namespace LinCms.Application.Selection.Services
{
    public class BIDashboardService
    {
        private readonly IAuditBaseRepository<ProductData> _productRepository;
        private readonly IAuditBaseRepository<StrategyExecution> _executionRepository;
        private readonly IAuditBaseRepository<RiskAlert> _alertRepository;

        public BIDashboardService(
            IAuditBaseRepository<ProductData> productRepository,
            IAuditBaseRepository<StrategyExecution> executionRepository,
            IAuditBaseRepository<RiskAlert> alertRepository)
        {
            _productRepository = productRepository;
            _executionRepository = executionRepository;
            _alertRepository = alertRepository;
        }

        public async Task<BIDashboardDto> GetDashboardDataAsync()
        {
            var products = await _productRepository.Select.ToListAsync();
            var executions = await _executionRepository.Select.Where(e => e.IsLatest).ToListAsync();
            var alerts = await _alertRepository.Select.OrderByDescending(a => a.CreateTime).Take(10).ToListAsync();

            return new BIDashboardDto
            {
                TotalProducts = products.Count,
                ActiveProducts = products.Count(p => p.Status == "active"),
                AverageScore = executions.Any() ? executions.Average(e => e.Score ?? 0) : 0,
                HighRiskCount = alerts.Count(a => a.Severity == "high"),
                TopProducts = products.Take(5).Select(p => new ProductKPIDto
                {
                    ProductId = p.Id,
                    ProductName = p.ProductName,
                    TotalScore = executions.Where(e => e.ProductId == p.Id).Average(e => e.Score),
                    EstimatedProfit = p.EstimatedMonthlySales * (p.TargetPrice - p.PurchaseCost),
                    RiskLevel = "low"
                }).ToList(),
                RecentAlerts = alerts.Select(a => new AlertDto
                {
                    Id = a.Id,
                    ProductId = a.ProductId,
                    AlertType = a.AlertType,
                    Severity = a.Severity,
                    Message = a.AlertMessage,
                    CreatedAt = a.CreateTime
                }).ToList()
            };
        }

        public async Task<ProductKPIDto> GetProductKPIAsync(long productId)
        {
            var product = await _productRepository.Select.Where(p => p.Id == productId).FirstAsync();
            if (product == null) return null;

            var executions = await _executionRepository.Select.Where(e => e.ProductId == productId && e.IsLatest).ToListAsync();

            return new ProductKPIDto
            {
                ProductId = product.Id,
                ProductName = product.ProductName,
                TotalScore = executions.Any() ? executions.Average(e => e.Score) : null,
                EstimatedProfit = product.EstimatedMonthlySales * (product.TargetPrice - product.PurchaseCost),
                ROI = product.TargetPrice > 0 ? (product.TargetPrice - product.PurchaseCost) / product.TargetPrice : null,
                RiskLevel = "low",
                StrategyCount = executions.Count
            };
        }

        public async Task<List<AlertDto>> GetAlertsAsync(int page = 1, int size = 20)
        {
            var alerts = await _alertRepository.Select
                .OrderByDescending(a => a.CreateTime)
                .Page(page, size)
                .ToListAsync();

            return alerts.Select(a => new AlertDto
            {
                Id = a.Id,
                ProductId = a.ProductId,
                AlertType = a.AlertType,
                Severity = a.Severity,
                Message = a.AlertMessage,
                CreatedAt = a.CreateTime
            }).ToList();
        }
    }
}
```

### 2. BI监控Controller (DashboardController.cs)

```csharp
using LinCms.Application.Selection.Services;
using Microsoft.AspNetCore.Mvc;
using System.Threading.Tasks;

namespace LinCms.Web.Controllers.Selection
{
    [Route("api/selection/dashboard")]
    [ApiController]
    public class DashboardController : ControllerBase
    {
        private readonly BIDashboardService _dashboardService;

        public DashboardController(BIDashboardService dashboardService)
        {
            _dashboardService = dashboardService;
        }

        [HttpGet("bi")]
        public async Task<IActionResult> GetBIDashboard()
        {
            var data = await _dashboardService.GetDashboardDataAsync();
            return Ok(new { code = 200, data });
        }

        [HttpGet("alerts")]
        public async Task<IActionResult> GetAlerts([FromQuery] int page = 1, [FromQuery] int size = 20)
        {
            var data = await _dashboardService.GetAlertsAsync(page, size);
            return Ok(new { code = 200, data });
        }
    }

    [Route("api/selection/products/{productId}")]
    [ApiController]
    public class ProductKPIController : ControllerBase
    {
        private readonly BIDashboardService _dashboardService;

        public ProductKPIController(BIDashboardService dashboardService)
        {
            _dashboardService = dashboardService;
        }

        [HttpGet("kpi")]
        public async Task<IActionResult> GetProductKPI(long productId)
        {
            var data = await _dashboardService.GetProductKPIAsync(productId);
            return Ok(new { code = 200, data });
        }
    }
}
```
